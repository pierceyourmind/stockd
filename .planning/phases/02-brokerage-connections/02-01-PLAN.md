```php
// Plaid helpers
function getPlaidClient(): \TomorrowIdeas\Plaid\Plaid {
    $env = $_ENV['PLAID_ENV'] ?? 'sandbox';
    return new \TomorrowIdeas\Plaid\Plaid(
        $_ENV['PLAID_CLIENT_ID'],
        $_ENV['PLAID_SECRET'],
        $env
    );
}

// Access token encryption (AES-256-GCM)
function encryptAccessToken(string $token): string {
    $key = $_ENV['ENCRYPTION_KEY'];
    if (empty($key)) {
        throw new \RuntimeException('ENCRYPTION_KEY not configured');
    }
    $key = hex2bin($key); // Convert hex string to binary
    $iv = random_bytes(12); // 96-bit IV for GCM
    $tag = '';
    $encrypted = openssl_encrypt($token, 'aes-256-gcm', $key, OPENSSL_RAW_DATA, $iv, $tag);
    if ($encrypted === false) {
        throw new \RuntimeException('Encryption failed');
    }
    // Combine IV + tag + ciphertext, base64 encode for storage
    return base64_encode($iv . $tag . $encrypted);
}

function decryptAccessToken(string $stored): string {
    $key = $_ENV['ENCRYPTION_KEY'];
    if (empty($key)) {
        throw new \RuntimeException('ENCRYPTION_KEY not configured');
    }
    $key = hex2bin($key);
    $data = base64_decode($stored);
    if ($data === false || strlen($data) < 28) { // 12 IV + 16 tag minimum
        throw new \RuntimeException('Invalid encrypted data');
    }
    $iv = substr($data, 0, 12);
    $tag = substr($data, 12, 16);
    $ciphertext = substr($data, 28);
    $decrypted = openssl_decrypt($ciphertext, 'aes-256-gcm', $key, OPENSSL_RAW_DATA, $iv, $tag);
    if ($decrypted === false) {
        throw new \RuntimeException('Decryption failed');
    }
    return $decrypted;
}
```
