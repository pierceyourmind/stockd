**2. Update Alpine.js methods in stockApp()**

a) Replace `connectBrokerage()` -- instead of a full-page redirect to SnapTrade, this now fetches a link_token and opens Plaid Link as an in-page modal:

```javascript
async connectBrokerage() {
    try {
        // Get link_token from backend
        const res = await fetch('api.php?action=plaidCreateLinkToken');
        const data = await res.json();

        if (data.error) {
            this.showToast(data.error, 'error');
            return;
        }

        // Open Plaid Link
        const handler = Plaid.create({
            token: data.link_token,
            onSuccess: async (publicToken, metadata) => {
                try {
                    // Exchange public_token for access_token via backend
                    const exchangeRes = await fetch('api.php?action=plaidExchangeToken', {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify({
                            public_token: publicToken,
                            institution: metadata.institution
                        })
                    });
                    const exchangeData = await exchangeRes.json();

                    if (exchangeData.success) {
                        this.showToast('Brokerage connected successfully!', 'success');
                        this.loadBrokerageConnections();
                    } else {
                        this.showToast(exchangeData.error || 'Failed to save connection', 'error');
                    }
                } catch (e) {
                    console.error('Token exchange failed:', e);
                    this.showToast('Failed to save connection. Please try again.', 'error');
                }
            },
            onExit: (err, metadata) => {
                if (err) {
                    console.error('Plaid Link error:', err);
                    this.showToast('Connection cancelled or failed. Please try again.', 'error');
                }
            }
        });

        handler.open();
    } catch (e) {
        console.error('Failed to start connection:', e);
        this.showToast('Failed to start brokerage connection. Please try again.', 'error');
    }
},
```

b) Replace `loadBrokerageConnections()` -- update fetch URLs from SnapTrade to Plaid endpoints:

```javascript
async loadBrokerageConnections() {
    this.connectionsLoading = true;
    try {
        const [connectionsRes, accountsRes] = await Promise.all([
            fetch('api.php?action=plaidConnections'),
            fetch('api.php?action=plaidAccounts')
        ]);
        const connectionsData = await connectionsRes.json();
        const accountsData = await accountsRes.json();

        if (connectionsData.connections) {
            this.brokerageConnections = connectionsData.connections;
        }
        if (accountsData.accounts) {
            this.brokerageAccounts = accountsData.accounts;
        }
    } catch (e) {
        console.error('Failed to load connections', e);
        this.showToast('Failed to load connections', 'error');
    }
    this.connectionsLoading = false;
},
```

c) Remove `reconnectBrokerage(connectionId)` method -- Plaid handles reconnection differently (through update mode link tokens). For now, just remove it. A reconnect feature can be added later if needed.
